async function loadTips(){ const list = document.getElementById("tipsList"); list.textContent = "Loading..."; try { const result = await request("/api/slips"); const slips = result.slips || []; list.innerHTML = ""; if (!slips.length) { list.textContent = "Hakuna FREE booking slips kwa sasa."; return; } slips.forEach(slip => { const card = document.createElement("div"); card.className = "tip"; const title = document.createElement("h3"); title.textContent = slip.title || "Booking Slip"; const odds = document.createElement("p"); odds.textContent = "Total Odds: " + (slip.totalOdds ?? "Not set"); const tipster = document.createElement("p"); tipster.textContent = "Tipster: " + (slip.tipster?.username || "Unknown"); card.append(title, odds, tipster); (slip.bookingCodes || []).forEach(code => { const p = document.createElement("p"); p.textContent = code.bookmaker + ": " + code.code; card.appendChild(p); }); list.appendChild(card); }); } catch (error) { list.textContent = error.message; } }
